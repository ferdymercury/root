//----- client flags
enum EObjBits {
   kCanDelete        = BIT(0),   // if object in a list can be deleted
   kObjIsPersistent  = BIT(2),   // if datamember is persistent (TDataMember)
   kObjIsParent      = BIT(1),   // if hyperlink is parent of linked list (TLink)
   kObjInCanvas      = BIT(3),   // if object has been inserted in a pad/canvas
   kModified         = BIT(5)    // if object has been modified
   kDoneByView       = BIT(6)    // if object was created by the TObjectView to back that
};

   //______________________________________________________________________________
   void TObjectView::AppendObject(Option_t *option)
   {
   //*-*-*-*-*-*-*-*-Append an image object to another TImageObject*-*-*-*-*
   //*-*             ==============================================
   //*-*  if an attempt to draw is made when a canvas is not yet created.
   //*-*  a default canvas is created with the name "c1".
   //*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*-*
      if (fgSelectedView) {
       SetBit(kObjInImage);
       fgSelectedView->Add(this,option);
       fgSelectedView->Modified(kTRUE);
     }
     else
       AppendPad(option);
   }

  //_____________________________________________
   TObjectView *TObjectView::Cd(){
   // return the previous selected object and set a current one
    TObjectView *current = fgSelectedView;
    fgSelectedView = this;
    return current;
   }
  //_____________________________________________
  TObjectView *TObjectView::DrawView()
  {
     AppendObject();
   //*-*
   //*-*  Draw the accessoures into this Image
   //*-*
     TObjectView *saveselected = Cd();
     Draw();
     saveselected->Cd();
  }

   //_____________________________________________
   void  TObjectView::RegisterObject(TObject *obj)
   {
     if (obj) {
    // We have to mark the TObject been created by ObjectImage with a
    // special bit
         obj->SetBit(kDoneByView);
    // Now we can Draw new attribute
         obj->Draw();
     }
   }

   //_____________________________________________
   void  TObjectView::Modified(Bool_t flag){
 //     TObject::Modified(flag);
      fModified = flag;
      fOriginView->Modified(flag);
   }


  //____________________________________________________
  void TObjectView::View(Int_t deepofview)
  {
    OwnView();
    if (deepofview) {
      if (deepofview > 0) deepofview--;   // decrease the current level
      TIter next(this);
      while (TObject *primitive = next())
                     primitive->View(deepofview);
    }
    Modified(kFALSE);
  }

  //____________________________________________________
  void TObjectView::Paint(Int_t deepofview)
  {
   View(deepofview);
  }
